\newcommand\mynoschapter[4]{%
  \clearpage\thispagestyle{mynosplain}%
  \vspace*{8mm}%
  {\fontsize{12}{15}\selectfont\bfseries\RaggedRight
    \ifx\relax#2\relax\else #2\ \fi #3\par}%
  \vspace{2mm}\hrule height .8pt\vspace{6mm}%
  \ifx\relax#1\relax\else\label{ch:#1}\fi
  \ifx\relax#1\relax
    \addcontentsline{toc}{chapter}{#4}%
  \else
    \addcontentsline{toc}{chapter}{\protect\numberline{#1.}#4}%
  \fi
  \markboth{\ifx\relax#2\relax\else #2 \fi #3}{\ifx\relax#2\relax\else #2 \fi #3}}
\newcommand\mynossection[3]{%

    \addcontentsline{toc}{section}{\protect\numberline{#1}#3}%
    \markright{#1 #2}\fi
  \nopagebreak\vspace{.2\baselineskip}\nopagebreak}
