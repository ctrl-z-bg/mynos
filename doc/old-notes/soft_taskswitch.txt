[On Fri, 15 Feb 2002, sefirot wrote:]
> hello,
> would you explain the way of software task-switching
> ( namely, switching without using tss ) on x86 in detail ?

 It's simple:
  -push all regs to the stack
  -load a new value into ss:esp (switch kernel stacks)
  -pop all regs from the new stack
  -load a new value into cr3 (pd base reg)
  -load a new ss:esp value into the system tss (patch it)
  -iret (when the task switcher is placed into an interrupt handler)

 OR even simplier: (faster; only for microkernels)
  -save all regs to the thread data struct
  -load all regs from the new thread data struct
  -reload cr3 (on process switches only)
  -iret

    Viktor

ps1:
 It can be as fast as 64 mov-s in case of a thread switch.
 (about 32 cycles on a P6 core /Ppro-PIII/)

 The 1st method modifies the tss. The x86 reads the ss0:esp values from
 the tss on a ring3->ring0 switch. This is the only required field.
 (Using one kernel stack per cpu saves the patch and the reload, but makes
 the kernel ring0 code uninterruptable. Actually... who wants to
 interrupt the task switcher in a critical section?)

// c scheduler.. "round robin" i think
void c_sched () {
asm ("mov %%esp, %0": "=r"(task_st[current_task].esp): );
task_st[current_task].esp+=16;

do {
current_task++;
if (current_task == max_tasks+1)
 current_task = 0;
} while (task_st[current_task].status != RUNNING);

asm ("mov %0, %%esp" :: "r" (task_st[current_task].esp));

asm ("popa");
outb (0x20, 0x20);
asm ("iret");
}

 the task struct...statically alloc'd stack 
struct {
char stack[20000];
unsigned long int esp;
char name[256];
byte status;
} task_st[10];

// the irq0 handler.
irq0_asm:
pushad
call c_sched ; call the C 'scheduler'

// struct used to 'push' values artificially.
struct da_stuff {
unsigned int regs[8];
unsigned int eip;
unsigned int cs;
unsigned int eflags;
unsigned int ret_eip; // this is for 'return' to work from tasks.
};

// init_task function
// just a init_task (&func_name, "this is a task!");
void init_task (unsigned long eip, char*name) {
struct da_stuff *da;
int num ;
num = max_tasks+1;
da = (struct da_stuff*)(task_st[num].stack +
sizeof(task_st[num].stack) - sizeof(struct da_stuff));
da->eip = eip;
da->cs = 0x8;
da->eflags = 0x202;
task_st[num].esp = (unsigned long)da;
strcpy1 (task_st[num].name, name);
task_st[num].status = RUNNING;
da->ret_eip = (unsigned int long)&exit_h;
max_tasks++;
}

void exit_h() {
task_st[current_task].status = STOPPED;
asm ("hlt"); // wait till next interrupt
}
